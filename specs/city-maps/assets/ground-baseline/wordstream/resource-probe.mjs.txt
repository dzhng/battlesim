import {createRequire} from 'node:module';
import {writeFile} from 'node:fs/promises';
import path from 'node:path';
const cwd=process.cwd(),require=createRequire(path.join(cwd,'web/package.json'));
const {chromium}=require('playwright'),{createServer}=await import(require.resolve('vite'));
const server=await createServer({root:path.join(cwd,'web'),configFile:path.join(cwd,'web/vite.config.ts'),server:{port:0,host:'127.0.0.1'},logLevel:'error'});
let browser;
try {
 await server.listen();
 browser=await chromium.launch({channel:'chrome',headless:true,args:['--enable-unsafe-webgpu','--enable-features=WebGPU','--use-angle=metal','--js-flags=--expose-gc']});
 const page=await browser.newPage({viewport:{width:1280,height:720}}),errors=[];
 page.on('response',r=>{if(r.status()>=400)console.error('HTTP',r.status(),r.url());});
 page.on('pageerror',e=>errors.push(String(e)));page.on('console',m=>{if(m.type()==='error'){errors.push(m.text());console.error(m.text());}});
 await page.goto(`http://127.0.0.1:${server.httpServer.address().port}/@fs/${cwd}/throwaway/city-spike/sa3/full-ground.html`);
 const cdp=await page.context().newCDPSession(page);
 const rows=[],cases=(process.env.SA3_CASES??'uniform').split(',');
 for(const mode of cases) for(const size of (process.env.SA3_SIZES??'12000,15000,18000').split(',').map(Number)) {
  const row=await page.evaluate(async({cwd,size,mode})=>{
   const file=p=>'/@fs/'+cwd+'/'+p;
   const [{createBattleFrame},{GroundView},{buildWorldLayers},{villageLight},{villageFogGeometry,villageFogStyle},{villageOverlayGlow,villagePaint},{villageModelDetail},{VILLAGE_RULES},biomeModule,wasm]=await Promise.all([
    import(file('packages/battle-renderer/src/frame/battleFrame.ts')),import(file('web/src/battle/sim/ground.ts')),import(file('packages/battle-renderer/src/worldMesh.ts')),
    import(file('apps/battle-lab/src/villageLight.ts')),import(file('apps/battle-lab/src/villageFog.ts')),import(file('apps/battle-lab/src/villageOverlay.ts')),import(file('apps/battle-lab/src/villageModels.ts')),import(file('apps/battle-lab/src/scenarios.ts')),import(file('fixtures/biomes/summer.json')),import('/src/wasm/game_wasm.js'),
   ]);
   const module=await wasm.default(),adapter=await navigator.gpu.requestAdapter({powerPreference:'high-performance'});
   if(adapter.info.vendor!=='apple'||adapter.isFallbackAdapter)throw new Error('Not Apple Metal');
   const device=await adapter.requestDevice(),validation=[];
   device.addEventListener('uncapturederror',e=>validation.push(e.error.message));
   const canvas=document.getElementById('probe'),context=canvas.getContext('webgpu'),format=navigator.gpu.getPreferredCanvasFormat();
   context.configure({device,format,alphaMode:'opaque'});
   const view=new wasm.WorldView(JSON.stringify({size:[size,size],height_grid_m:4,fog_cell_m:8,slope_cutoff_deg:35}),JSON.stringify(VILLAGE_RULES));
   const exports={terrain:{...JSON.parse(view.terrain_grid()),pageIds:view.terrain_page_ids(),heights:view.terrain_heights()},positions:view.terrain_positions(),indices:view.terrain_indices(),triangleSurfaces:view.terrain_triangle_surfaces(),props:view.props(),water:view.water(),forests:view.forests(),foliage:view.foliage(),roads:view.roads()};
   const world=buildWorldLayers(exports,JSON.parse(wasm.world_layout(JSON.stringify(VILLAGE_RULES))),biomeModule.default,'surface');
   const setupStart=performance.now();
   const scene=await createBattleFrame(device,format,{light:villageLight,fogGeometry:villageFogGeometry,fogStyle:villageFogStyle,overlayGlow:villageOverlayGlow,paint:villagePaint,models:villageModelDetail,world,instances:[],width:1280,height:720});
   const frameSetupMs=performance.now()-setupStart;
   const ground=new GroundView({cellM:1,cols:size,rows:size}),tiles=Math.ceil(size/16),hole=mode==='mixed'?Math.floor(tiles/2)*tiles+Math.floor(tiles/2):-1;
   let count=0;for(let y=0;y<tiles;y++)for(let x=0;x<tiles;x++)count+=Math.min(16,size-x*16)===16?1:Math.min(16,size-y*16);
   let runs=new Float32Array(count*4),at=0;
   for(let y=0;y<tiles;y++)for(let x=0;x<tiles;x++){
    const width=Math.min(16,size-x*16),height=Math.min(16,size-y*16),tile=y*tiles+x;
    if(tile===hole)continue;
    // Saturated clearing and exact raw native tracks zero. The established
    // observation projection presents tracks=max(raw,cleared)=255.
    if(width===16){runs.set([tile,height*16*256,0,255*65536],at);at+=4;}
    else for(let row=0;row<height;row++){runs.set([tile,row*16+width*256,0,255*65536],at);at+=4;}
   }
   const before=performance.memory?.usedJSHeapSize??null,start=performance.now();
   ground.applyRuns({epoch:1,side:'blue',baseRevision:0,revision:1,full:true,runs:runs.subarray(0,at)});
   const applyMs=performance.now()-start,inputBytes=at*4; runs=null;
   if(ground.clearedCount!==size*size-(hole>=0?256:0)||ground.at(size-.5,size-.5).tracks!==255)throw new Error('Lost full learned clearing');
   let localApplyMs=0;const localValues=[];
   if(mode!=='uniform'){
    const points=mode==='fragmented'?Array.from({length:20000},(_,k)=>[Math.floor((k%100+.5)*size/100),Math.floor((Math.floor(k/100)+.5)*size/200)]):[[0,0],[15,15],[16,16],[1023,1023],[1024,1024],[1025,1024],[size-1,size-1],[size-17,size-17]];
    const packed=points.filter(([x,y])=>x>=0&&y>=0&&x<size&&y<size).map(([x,y],k)=>({tile:Math.floor(y/16)*tiles+Math.floor(x/16),local:y%16*16+x%16,a:91+k%100+(77+k%100)*256,b:28+(67+k%100)*256+255*65536,x,y})).sort((a,b)=>a.tile-b.tile||a.local-b.local);
    const delta=Float32Array.from(packed.flatMap(p=>[p.tile,p.local+256,p.a,p.b]));const t=performance.now();ground.applyRuns({epoch:1,side:'blue',baseRevision:1,revision:2,full:false,runs:delta});localApplyMs=performance.now()-t;
    for(const p of packed){const got=ground.cell(p.x,p.y),want={crater:p.a&255,scorch:p.a>>>8,tracks:255,trampled:p.b>>>8&255};if(JSON.stringify(got)!==JSON.stringify(want))throw new Error('Lost local learned channels');}localValues.push(packed.length);
   }
   const receiver={localApplyMs,localValues,payloadBytes:ground.byteLength,clearedCells:ground.clearedCount,hasMarks:ground.hasMarks,applyMs,inputBytes,heapBefore:before,heapAfter:performance.memory?.usedJSHeapSize??null};
   scene.setGround(ground);
   const phases=[];
   const draw=async(name,camera)=>{
    const before=scene.stats().scars.totalBytes,t=performance.now();
    scene.render(context.getCurrentTexture().createView(),{camera3d:camera,width:1280,height:720});
    const cpuMs=performance.now()-t;await device.queue.onSubmittedWorkDone();
    phases.push({name,cpuMs,completeMs:performance.now()-t,uploadedBytes:scene.stats().scars.totalBytes-before,scars:scene.stats().scars,memory:scene.stats().memory});
   };
   const lens={fovY:.78,aspect:1280/720,near:.1};
   const tactical={target:[1024,1024,0],distance:150,pitch:.8,yaw:0,...lens};
   const overview={target:[size/2,size/2,0],distance:size*1.7,pitch:1.4,yaw:0,...lens};
   await draw('tactical',tactical);await draw('tactical-unchanged',tactical);
   for(let k=0;k<32;k++)await draw('tactical-steady-'+k,tactical);
   await draw('overview',overview);await draw('overview-unchanged',overview);
   for(let k=0;k<32;k++)await draw('overview-steady-'+k,overview);
   window.__sa3Shot=canvas.toDataURL();
   scene.setGround(null);scene.dispose();view.free();context.unconfigure();await device.queue.onSubmittedWorkDone();
   const finalCounts=scene.stats().memory;device.destroy();
   if(validation.length)throw new Error(validation.join('\n'));
   return {size,mode,frameSetupMs,gpu:{vendor:adapter.info.vendor,architecture:adapter.info.architecture,maxTextureDimension2D:device.limits.maxTextureDimension2D},receiver,phases,wasmHighWater:module.memory.buffer.byteLength,finalCounts,validation};
  },{cwd,size,mode});
  rows.push(row);const shot=await page.evaluate(()=>window.__sa3Shot);await writeFile(`throwaway/city-spike/sa3/overview-${mode}-${size}.png`,Buffer.from(shot.split(",")[1],"base64"));
  await cdp.send('HeapProfiler.collectGarbage');
  await writeFile(`throwaway/city-spike/sa3/${process.env.SA3_OUT??'full-ground'}.json`,JSON.stringify({rows,errors},null,2)+'\n');
  console.log(JSON.stringify(row));
 }
 if(errors.length)throw new Error(errors.join('\n'));
}finally {await browser?.close();await server.close();}
